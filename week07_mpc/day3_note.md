# 第一部分 什么是滚动优化
今天新学的四行代码要刻在脑子里：
p, v = 0.0, 0.0                # 真实状态
for k in range(总拍数):    
    U_best = 解QP(当前 p, v)    # ① 往前算 N 步  用 cvxpy 算出一条完美的未来轨迹和油门序列
    a0 = U_best[0]             # ② 【只取第一步】  把算出来的最优油门序列里的第一个拿出来执行
    真实系统走一步(a0)          # ③ 世界前进   把这个 a0 给小车的真实物理模型，往前走一步
    p, v = 新状态               # ④ 下一拍用【实测状态】重来   用小车走完这一步之后的【真实状态】，重新回到第 1 步去算

它和 Day 2 的差别是本质的：
 
             Day 2	                         Day 3
解几次 QP	1 次（全部走完）	            每一拍都解一次
用什么状态	假设的 p0=0	                 上一拍真实走出来的
执行几步	只是看答案	                   只执行第一步

算 N 步、只做 1 步、下一拍全部推翻重算 —— 这叫「滚动」
如果一次算完 N 步全部执行，那就不叫 MPC 了，叫开环规划：模型一有误差就一路错下去，没有任何修正机会。

为什么叫「反馈校正」
你每拍用真实状态重新解 QP，等于每拍都在问：「现在实际在这儿，接下来怎么走？」
误差不会累积，因为每一拍都把误差吃掉了。 —— 这就是 MPC 比开环规划强的地方

# 第二部分 闭环主循环的框架

p, v 是真实世界里小车的状态。
ps[k], vs[k] 是用 cvxpy 预测出来的状态。

在 mpc_step 函数里，用的是 ps 和 vs（预测）。但到了主循环走完一步后，必须用真实的物理公式更新 p 和 v

## 参数
dt      = 0.1       # 采样周期
u_lim   = 2.0       # |a| ≤ 2，执行器饱和
Q       = 1.0       # 位置误差权重
R       = 0.1       # 控制量权重
horizon = 10        # 预测时域 N（见day1 note 与控制时域Nc对比）
p_ref   = 10.0      # 目标位置
steps   = 120       # 跑 120 拍 = 12 秒

## 每一拍的QP
def mpc_step(p, v, p_ref, N):    # p（位置）、v（速度）、目标位置 p_ref 和预测步数 N
    """解一拍 MPC，返回第一步控制量"""
    a = cp.Variable(N)    # 决策变量：未来 N 步的加速度  用 cvxpy 声明一个长度为 N 的一维数组。这是我们要让求解器去寻找的未知数

    ps, vs = [p], [v]     # 用表达式递推预测  创建两个列表，把当前真实状态（第 0 步）塞进去，作为预测的起点
    for k in range(N):
        ps.append(ps[k] + dt*vs[k] + (dt**2/2)*a[k])
        vs.append(vs[k] + dt*a[k])
    
    cost = sum(Q * cp.square(ps[k] - p_ref) for k in range(1, N+1))   
    # 把未来每一步预测的位置 ps[k] 减去目标 p_ref 求平方（误差），乘上权重 Q，然后全部加起来
    cost += R * cp.sum_squares(a)
    # 把未来所有控制量 a 求平方和，乘上权重 R，惩罚用力过猛

    cons = [a >= -u_lim, a <= u_lim]   # 物理约束。电机只能给出这么大力
    prob = cp.Problem(cp.Minimize(cost), cons)
    prob.solve()

    if prob.status != "optimal":
        return None
    return float(a.value[0])     # 只取第一个

## 闭环主循环
p, v = 0.0, 0.0              # 真实状态（小车）
hist_p, hist_v, hist_a = [], [], []

for k in range(steps):
    a0 = mpc_step(p, v, p_ref, horizon)    # 进入时间循环。把当前的真实状态 p, v 喂给 mpc_step 函数

    hist_p.append(p)
    hist_v.append(v)
    hist_a.append(a0)
    # 以上三行记录当前状态。这还没更新，记录的是执行这脚油门之前的状态

    # 真实系统用【和模型一样】的方程往前走
    p = p + dt*v + (dt**2/2)*a0
    v = v + dt*a0
    # 把算出来的 a0 应用到物理引擎（或真实世界）。根据物理公式，得出下一拍的真实位置和真实速度
    # 这里绝对不能写成 p = ps[1]。因为 ps[1] 是模型预测出来的“理想状态”。而在现实中，有摩擦力、有风、有模型误差。每一次循环，都要用真实更新的 p, v，这样到了下一拍 mpc_step 时，才能根据真实的偏差去修正轨迹。如果写成 p = ps[1]，那就是“开环”，现实早就跑偏了
    
    # 在真实的机器人 C++ 代码里，这行代码实际上长这样：
    p_new = read_encoder() // 从传感器读取轮子实际转了多少
    v_new = read_imu() // 从传感器读取实际速度
    在 Python 仿真里，没法读传感器，所以用 “上一拍的真实状态 p + 当前真实执行的油门 a0” 来推算出这一拍的真实状态
